#!/usr/bin/env python3
"""Extract one steady-state training step's timeline from an nsys sqlite export.

Answers, from the trace rather than from configuration:
  - when the first gradient AllReduce starts, relative to backward start
  - how much of backward runs with a collective resident
  - when the last AllReduce ends, relative to backward end (the exposed tail)

Usage: p9_timeline.py <report.sqlite> [device_id]
"""
import sqlite3, sys, json

db, dev = sys.argv[1], int(sys.argv[2]) if len(sys.argv) > 2 else 0
c = sqlite3.connect(db)

def has(t):
    return c.execute("SELECT name FROM sqlite_master WHERE type='table' AND name=?",
                     (t,)).fetchone() is not None

# --- NVTX 'backward' ranges, steady state only -----------------------------
nv_all = c.execute("""
    SELECT e.start, e.end, e.globalTid
    FROM NVTX_EVENTS e LEFT JOIN StringIds s ON s.id = e.textId
    WHERE COALESCE(s.value, e.text) = 'backward'
    ORDER BY e.globalTid, e.start
""").fetchall()
# One process only: ranges from different ranks interleave in time, and a
# window bounded by another rank's next backward would be meaningless.
tid = nv_all[0][2] if nv_all else None
nv = [(a, b) for a, b, t in nv_all if t == tid]
if not nv:
    print(json.dumps({"error": "no NVTX 'backward' ranges"})); sys.exit(1)

# one NVTX range per rank per step; take a middle one to avoid warmup effects
i = len(nv) // 2
bwd_start, bwd_end = nv[i][0], nv[i][1]
# Bound the window at the NEXT backward on this rank, so the tail cannot be
# contaminated by the following step's collectives.
next_start = nv[i + 1][0] if i + 1 < len(nv) else bwd_end + 200_000_000

kern_tbl = "CUPTI_ACTIVITY_KIND_KERNEL"
if not has(kern_tbl):
    print(json.dumps({"error": f"no {kern_tbl}"})); sys.exit(1)

# window: the backward range plus a generous tail so trailing collectives show
win_end = next_start
rows = c.execute(f"""
    SELECT k.start, k.end, s.value, k.deviceId
    FROM {kern_tbl} k JOIN StringIds s ON s.id = k.shortName
    WHERE k.deviceId = ? AND k.end > ? AND k.start < ?
    ORDER BY k.start
""", (dev, bwd_start, win_end)).fetchall()

nccl = [r for r in rows if "nccl" in r[2].lower()]
comp = [r for r in rows if "nccl" not in r[2].lower() and r[0] < bwd_end]

def ms(ns):
    return round(ns / 1e6, 3)

out = {
    "device": dev,
    "backward_span_ms": ms(bwd_end - bwd_start),
    "nvtx_backward_ranges_this_rank": len(nv),
    "window_end_ms_after_backward_end": ms(next_start - bwd_end),
    "nccl_kernels_in_window": len(nccl),
}
if nccl:
    first, last = nccl[0], max(nccl, key=lambda r: r[1])
    out["first_allreduce_start_ms_after_backward_start"] = ms(first[0] - bwd_start)
    out["first_allreduce_start_pct_of_backward"] = round(
        100.0 * (first[0] - bwd_start) / max(bwd_end - bwd_start, 1), 1)
    out["last_allreduce_end_ms_after_backward_end"] = ms(last[1] - bwd_end)
    out["nccl_busy_ms_total"] = ms(sum(r[1] - r[0] for r in nccl))
    # union of NCCL intervals intersected with the backward compute span
    iv, cur_s, cur_e = [], None, None
    for s, e, _n, _d in sorted(nccl):
        if cur_s is None:
            cur_s, cur_e = s, e
        elif s <= cur_e:
            cur_e = max(cur_e, e)
        else:
            iv.append((cur_s, cur_e)); cur_s, cur_e = s, e
    if cur_s is not None:
        iv.append((cur_s, cur_e))
    inside = sum(max(0, min(e, bwd_end) - max(s, bwd_start)) for s, e in iv)
    out["nccl_resident_during_backward_ms"] = ms(inside)
    out["nccl_resident_pct_of_backward"] = round(
        100.0 * inside / max(bwd_end - bwd_start, 1), 1)
    out["nccl_after_backward_ms"] = ms(sum(max(0, e - max(s, bwd_end)) for s, e in iv))
    out["allreduce_durations_ms"] = [ms(r[1] - r[0]) for r in nccl]
if comp:
    out["backward_kernels_in_span"] = len(comp)
print(json.dumps(out, indent=2))
