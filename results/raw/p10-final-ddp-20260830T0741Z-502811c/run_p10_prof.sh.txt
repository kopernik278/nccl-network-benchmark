#!/bin/bash
# Phase 10 timeline traces. Confirmation of an already-established mechanism,
# not a new investigation. Profiled runs are never timing samples.
set -u
cd /root/repo
OUT=/root/p10_out/prof; mkdir -p "$OUT"
T=src/ddp/train_ddp.py
M='--layers 8 --heads 12 --embd 768 --vocab 16384 --seq 1024 --batch 16 --pool 4 --precision bf16'
TRANSPORT=$(cat /root/p10_out/transport_env.txt 2>/dev/null)
# shellcheck disable=SC2086
export ${TRANSPORT:-DUMMY_UNUSED=1}
NS=$(ls /opt/nvidia/nsight-systems/*/target-linux-x64/nsys | head -1)
echo "nsys=$NS  transport=${TRANSPORT:-<NCCL defaults>}"

for MB in 25 4 64; do
  tag="final-${MB}mb"
  timeout -k 20 900 "$NS" profile -t cuda,nvtx --force-overwrite true \
    --cuda-memory-usage=false -o "$OUT/$tag" \
    torchrun --nproc_per_node=4 --master_port=$((29810 + MB % 10)) $T \
      --mode ddp --bucket-mb $MB --warmup 6 --steps 6 --repeats 1 --nvtx $M \
    > "$OUT/$tag.run.txt" 2>&1
  echo "$tag rc=$?"
  "$NS" stats --force-export=true --report cuda_gpu_kern_sum --format csv \
    "$OUT/$tag.nsys-rep" 2>/dev/null > "$OUT/$tag.cuda_gpu_kern_sum.csv"
  "$NS" stats --report nvtx_sum --format csv \
    "$OUT/$tag.nsys-rep" 2>/dev/null > "$OUT/$tag.nvtx_sum.csv"
  python3 /root/p10_timeline.py "$OUT/$tag.sqlite" 0 > "$OUT/$tag.timeline.json" 2>&1
  python3 - "$OUT/$tag.timeline.json" <<'PY'
import json, sys
d = json.load(open(sys.argv[1])); d.pop("allreduce_durations_ms", None)
print("   " + json.dumps(d))
PY
done
echo "===== PROF DONE ====="
